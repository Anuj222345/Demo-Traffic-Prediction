train.py 
from ultralytics import YOLO

def train_model():
    # Load YOLOv8 nano pre-trained weights
    model = YOLO('yolov8n.pt')
    
    # Train on Self-Driving Cars v6 Dataset (15 Classes)
    results = model.train(
        data='data.yaml',
        epochs=50,
        imgsz=416,
        batch=16,
        device=0, # GPU index or 'cpu'
        name='traffic_signs_v6_run'
    )
    print('Training completed successfully! Weights saved to runs/detect/traffic_signs_v6_run/weights/best.pt')

if __name__ == '__main__':
    train_model()
